Production Cloudflare Email Worker Code (worker.js)

Deploy this lightweight worker to your Cloudflare account to handle the edge email stream and KV store. 

export default {
  async email(message, env, ctx) {
    // 1. Extract the clean student prefix
    const recipient = message.to.split('@')[0].toLowerCase();
    
    // 2. Fetch destination Gmail from Cloudflare KV Store
    let targetGmail = await env.STUDENTS_KV.get(recipient);
    
    // Fallback: If not mapped individually, route to primary admin cluster
    if (!targetGmail) {
      targetGmail = env.FALLBACK_GMAIL || "cluster1@gmail.com";
    }

    // 3. Deliver natively via Cloudflare Edge Routing
    try {
      await message.forward(targetGmail);
      console.log(`[SUCCESS] Forwarded ${message.to} -> ${targetGmail}`);
    } catch (err) {
      console.error(`[ERROR] Forward failed: ${err.message}`);
    }
  }
};
